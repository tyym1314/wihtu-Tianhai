 let y=jt>=0&&jt<1?4*1.2*jt*(1-jt):0,distance=Math.min(TRACK_LENGTH,p.distance+p.speed*clamp(elapsed,0,250)/1000),scale=1;
 if(p.fallUntilTick!==null){const ft=clamp((tick-p.fallStartTick)/90,0,1);y=-Math.min(2.4,ft*5);scale=Math.max(.15,1-ft*.9);distance=p.distance;}return{x:mix(LANES[p.fromLane],LANES[p.lane],lt),y,distance,tick,scale};}
