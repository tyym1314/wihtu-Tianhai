import * as T from'three';
import runUrl from'./assets/atlases/run-back.webp?url';
import actionUrl from'./assets/atlases/action-back.webp?url';
import frontUrl from'./assets/atlases/front-win.webp?url';
export const RUN_ATLAS_URLS={run:runUrl,action:actionUrl,front:frontUrl};
const order=['fox','cat','bear','rabbit'],loader=new T.TextureLoader();let cache;
function atlases(){return cache??=Promise.all(Object.entries(RUN_ATLAS_URLS).map(async([key,url])=>{const t=await loader.loadAsync(url);t.colorSpace=T.SRGBColorSpace;t.magFilter=T.LinearFilter;t.minFilter=T.LinearFilter;t.generateMipmaps=false;return[key,t];})).then(p=>Object.fromEntries(p));}
/** High-detail fixed-angle original art on true world-positioned depth-tested sprites.
 * This is deliberately 2.5D presentation. It never computes position, collision or results. */
export function createIllustratedRunner(kind='fox',ghost=false,{onError=()=>{}}={}){
 const row=Math.max(0,order.indexOf(kind)),root=new T.Group(),sprite=new T.Sprite(new T.SpriteMaterial({transparent:true,alphaTest:.12,depthTest:true,depthWrite:false,toneMapped:false}));sprite.center.set(.5,.035);sprite.scale.set(1.65,1.65,1);root.add(sprite);sprite.visible=false;
 const shadow=new T.Mesh(new T.CircleGeometry(.38,32),new T.MeshBasicMaterial({color:0x342b19,transparent:true,opacity:.18,depthWrite:false}));shadow.rotation.x=-Math.PI/2;shadow.position.y=.012;root.add(shadow);
 const ring=new T.Mesh(new T.TorusGeometry(.42,.016,6,36),new T.MeshBasicMaterial({color:ghost?0x64bbb4:0xffd47b,transparent:true,opacity:ghost?.32:.75}));ring.rotation.x=Math.PI/2;ring.position.y=.018;root.add(ring);
 const shield=new T.Mesh(new T.SphereGeometry(.85,24,18),new T.MeshPhysicalMaterial({color:0x65fff1,transparent:true,opacity:.09,roughness:.1,depthWrite:false}));shield.position.y=.78;shield.visible=false;root.add(shield);
 root.userData={character:kind,loaded:false,bones:0,clip:'loading',modelSource:'Illustrated atlas on authoritative 3D trajectory',renderer:'atlas',trophy:'embedded'};
 let textures=null,dead=false,lastArgs=null,lastState=null,recoverTick=-100,finishVisual=null,lastSheet=null,lastColumn=-1;
 const ready=atlases().then(source=>{if(dead)return;textures=Object.fromEntries(Object.entries(source).map(([k,t])=>{const c=t.clone();c.repeat.set(.25,.25);c.needsUpdate=true;return[k,c];}));root.userData.loaded=true;sprite.visible=true;if(lastArgs)pose(...lastArgs);else pose('idle',0,0);}).catch(onError);
 function frame(sheet,column){if(!textures)return;if(sheet!==lastSheet){sprite.material.map=textures[sheet];sprite.material.needsUpdate=true;lastSheet=sheet;}if(column!==lastColumn||sheet!==root.userData.sheet){textures[sheet].offset.set(column/4,1-(row+1)/4);lastColumn=column;}root.userData.sheet=sheet;root.userData.frame=column;root.userData.uv=textures[sheet].offset.toArray();}
 function pose(clip,elapsed,time=elapsed){lastArgs=[clip,elapsed,time];let sheet='front',column=3,bob=0,tilt=0,sy=1;
  if(clip==='run'){sheet='run';column=Math.floor(Math.max(0,elapsed)*5)%4;bob=.025*(1-Math.cos(elapsed*Math.PI*5));tilt=Math.sin(elapsed*Math.PI*2.5)*.018;}
  else if(['jump','land','slide','hit','fall','recover'].includes(clip)){sheet='action';column=clip==='slide'?1:clip==='hit'?2:clip==='fall'?3:0;if(clip==='slide')sy=.78;if(clip==='fall')tilt=elapsed*1.8;if(clip==='land'){sheet='run';column=0;sy=1-Math.sin(Math.min(1,elapsed/.33)*Math.PI)*.10;}if(clip==='recover'){sheet='run';column=0;}if(clip==='hit')tilt=Math.sin(elapsed*24)*.06;}
  else if(clip==='finish'){sheet='front';column=0;bob=Math.sin(Math.min(1,elapsed/1.2)*Math.PI)*.06;}
  else if(clip==='celebrate'){sheet='front';column=Math.floor(Math.max(0,elapsed)/1.3)%2+1;bob=Math.sin(elapsed*3)*.025;tilt=Math.sin(elapsed*2)*.015;}
  frame(sheet,column);sprite.position.y=bob;sprite.material.rotation=tilt;sprite.scale.set(1.65,1.65*sy,1);root.userData.clip=clip;root.userData.clipTime=elapsed;
 }
 function animate(p,tick,time){const previous=lastState;if(previous?.fallUntilTick!==null&&previous&&p.fallUntilTick===null)recoverTick=tick;lastState={...p};let clip='run',elapsed=p.distance/10;
  if(p.finishTick!==null){if(finishVisual===null)finishVisual=time-Math.max(0,(tick-p.finishTick)/60);const t=time-finishVisual;clip=t<1.2?'finish':p.visualWinner===false?'idle':'celebrate';elapsed=t<1.2?t:t-1.2;}
  else if(p.fallUntilTick!==null){clip='fall';elapsed=(tick-p.fallStartTick)/60;}
  else if(tick<p.stunUntilTick){clip='hit';elapsed=(tick-(p.stunUntilTick-48))/60;}
  else if(tick-recoverTick<36&&recoverTick>=0){clip='recover';elapsed=(tick-recoverTick)/60;}
  else if(tick<p.slideUntilTick){clip='slide';elapsed=(tick-(p.slideUntilTick-36))/60;}
  else if(p.jumpStartTick!==null&&tick-p.jumpStartTick<48){clip='jump';elapsed=(tick-p.jumpStartTick)/60;}
  else if(p.jumpStartTick!==null&&tick-p.jumpStartTick<58){clip='land';elapsed=(tick-p.jumpStartTick-48)/60;}
  // A waiting/stunned racing animal remains rear-facing until the actual finish.
  else if(p.speed===0){clip='run';elapsed=0;}
  pose(clip,elapsed,time);shield.visible=tick<p.shieldUntilTick;shadow.visible=p.fallUntilTick===null;ring.visible=shadow.visible;root.userData.authoritativeTick=tick;
 }
 return{root,ready,pose,animate,getBone(){return null},dispose(){dead=true;for(const t of Object.values(textures||{}))t.dispose();for(const m of[sprite,ring,shadow,shield]){if(m!==sprite)m.geometry?.dispose();m.material.dispose();}root.clear();}};
}
